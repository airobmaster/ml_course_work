def style_ax(ax, hide_spines=("top", "right", "left")):
    for s in hide_spines:
        ax.spines[s].set_visible(False)
    ax.spines["bottom"].set_color(BASELINE)


def bar_labels(ax, bars, fmt="{:,.0f}", color=INK_SECONDARY, pad=3, horizontal=False):
    for b in bars:
        if horizontal:
            w = b.get_width()
            ax.text(w + pad, b.get_y() + b.get_height() / 2, fmt.format(w),
                     va="center", ha="left", fontsize=9.5, color=color)
        else:
            h = b.get_height()
            ax.text(b.get_x() + b.get_width() / 2, h + pad, fmt.format(h),
                     va="bottom", ha="center", fontsize=9.5, color=color)


# ---------------------------------------------------------------
# Section 4: Target distribution (ordinal severity -> sequential blue ramp)
# ---------------------------------------------------------------
counts = train_labels[TARGET_COL].value_counts().sort_index()

fig, ax = plt.subplots(figsize=(6.5, 4.2))

    ax.text(b.get_x() + b.get_width() / 2, b.get_height() + 2500,
            f"{b.get_height():,.0f}\n({p:.1f}%)", ha="center", va="bottom",
            fontsize=10, color=INK_SECONDARY)
ax.set_ylabel("Number of buildings")
ax.set_title("Target Distribution: damage_grade (train)", color=INK_PRIMARY, fontsize=12, loc="left", pad=12)
ax.set_ylim(0, counts.max() * 1.22)
ax.yaxis.set_major_formatter(mticker.FuncFormatter(lambda x, _: f"{int(x):,}"))

ax.grid(axis="x", visible=False)
plt.tight_layout()

plt.close()

# ---------------------------------------------------------------
# Section 9: damage_grade by foundation_type (100% stacked, ordinal severity -> seq blue ramp)
# ---------------------------------------------------------------
ct = pd.crosstab(df["foundation_type"], df[TARGET_COL], normalize="index").sort_index()
order = ct.sum(axis=1).index  # keep natural order h,i,r,u,w
fig, ax = plt.subplots(figsize=(8, 4.8))
left = np.zeros(len(ct))
grade_colors = {1: SEQ_BLUE[250], 2: SEQ_BLUE[450], 3: SEQ_BLUE[650]}
for grade in [1, 2, 3]:
    vals = ct[grade].values * 100
    bars = ax.barh(ct.index, vals, left=left, color=grade_colors[grade], height=0.6,
                    label=f"Grade {grade}", edgecolor=SURFACE, linewidth=1)
    for y, (l, v) in enumerate(zip(left, vals)):
        if v > 5:
            ax.text(l + v / 2, y, f"{v:.0f}%", ha="center", va="center", fontsize=9,
                    color="white" if grade in (2, 3) else INK_PRIMARY)
    left += vals
ax.set_xlim(0, 100)
ax.set_xlabel("Share of buildings (%)")
ax.set_title("damage_grade Distribution by foundation_type", color=INK_PRIMARY, fontsize=12, loc="left", pad=12)
ax.legend(loc="upper center", bbox_to_anchor=(0.5, -0.18), ncol=3, frameon=False, fontsize=9.5)
style_ax(ax, hide_spines=("top", "right", "left"))
ax.grid(axis="y", visible=False)
plt.tight_layout()
plt.savefig(OUT / "damage_by_foundation.png", dpi=160)
plt.close()

# ---------------------------------------------------------------
# Section 9: correlation with target (diverging blue<->red, polarity)
# ---------------------------------------------------------------
corr = df[NUMERIC_COLS + [TARGET_COL]].corr()[TARGET_COL].drop(TARGET_COL).sort_values()
colors_corr = [DIVERGE_RED if v < 0 else DIVERGE_BLUE for v in corr.values]
fig, ax = plt.subplots(figsize=(8.5, 4.8))
bars = ax.barh(corr.index, corr.values, color=colors_corr, height=0.6)
pad = 0.012
for b, v in zip(bars, corr.values):
    ax.text(v + (pad if v >= 0 else -pad), b.get_y() + b.get_height() / 2, f"{v:+.3f}",
            va="center", ha="left" if v >= 0 else "right", fontsize=9, color=INK_SECONDARY)
ax.axvline(0, color=BASELINE, linewidth=1)
ax.set_xlim(corr.min() - 0.05, corr.max() + 0.05)
ax.set_xlabel("Pearson correlation with damage_grade")
ax.set_title("Numeric Feature Correlation with damage_grade", color=INK_PRIMARY, fontsize=12, loc="left", pad=12)
style_ax(ax, hide_spines=("top", "right", "left"))
ax.grid(axis="y", visible=False)
plt.tight_layout()
